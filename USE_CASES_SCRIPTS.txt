# Scripts de diagrammes de cas d'utilisation (PlantUML)
# Application Tech Info Plus - état fonctionnel Novembre 2025

## 1. Vue globale - Authentification & rôles
@startuml
left to right direction
skinparam actorStyle awesome
actor "Visiteur" as Visitor
actor "Utilisateur" as User
actor "Administrateur" as Admin
actor "Comptable" as Accountant
actor "Vendeur" as Seller
actor "Caissier" as Cashier

rectangle "Plateforme Tech Info Plus" {
  usecase UC_Login as "Se connecter"
  usecase UC_Logout as "Se déconnecter"
  usecase UC_ViewDashboard as "Consulter dashboard"
  usecase UC_UpdateProfile as "Mettre à jour profil"
}

Visitor --> UC_Login
User --> UC_Logout
User --> UC_ViewDashboard
User --> UC_UpdateProfile
Admin --> UC_ViewDashboard
Admin --> UC_UpdateProfile
Accountant --> UC_ViewDashboard
Seller --> UC_ViewDashboard
Cashier --> UC_ViewDashboard

UC_ViewDashboard .> UC_Login : <<include>>
@enduml

  usecase UC_CreateUser as "Créer un utilisateur"
  usecase UC_EditUser as "Modifier un utilisateur"
  usecase UC_DeactivateUser as "Activer / désactiver"
  usecase UC_AssignRole as "Assigner un rôle (Admin/Comptable/Vendeur/Caissier)"

Admin --> UC_AssignRole
Admin --> UC_SetPermissions

@enduml
